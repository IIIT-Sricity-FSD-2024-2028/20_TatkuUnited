User [icon: user, color: blue] {
  id string pk
  name string
  email string unique
  password string // password as hash
  gender enum
  phone string unique
  dob date
  
  role enum // [customer, provider, manager, admin]
  status enum // [active, blocked]
  
  cart {address: string, items: [{service: ref:Service, scheduledDate: date, startTime: time, endTime: time}]}
  addresses [{id: string, label: string, line: string, city: string, location: 2dsphere}]
}

Provider [icon: person, color: purple] {
  id string pk
  user ref:User fk
  offeredServices [ref:Service]
  experience number
  
  workHourStart number // minutes from midnight
  workHourEnd number // minutes from midnight
  workingDays [number] 
  
  location 2dsphere
  status enum // [pending, approved, suspended]
  
  rating number
  ratingCount number
  region ref:Region

  razorpayAccountId string
}

Manager [icon: person-standing, color: purple] {
  id string pk
  user ref:User fk
  salary number // integer in paise
  razorpayAccountId string
  
  experience number
  location 2dsphere
  status enum // [active, suspended, inactive]
}

Category [icon: option] {
  id string pk
  name string
  description string

  imageUrl string
  isActive bool
}

Service [icon: service, color: green] {
  id string pk
  name string
  category ref:Category fk
  
  reviewCount number
  rating number
  
  description string
  price number // integer in paise
  duration number
  isActive bool
  
  imageUrls [string]
  whatIsCovered [string]
  whatIsNotCovered [string]
  
  howItWorks [{title: string, desc: string}]
  faq [{question: string, answer: string}]
}

Order [icon: shopping-cart, color: orange] {
  id string pk
  customer ref:User fk
  address {line: string, city: string, location: 2dsphere} // snapshot

  totalAmount number // paise
  payment ref:Payment fk

  paymentStatus enum // [created, paid, failed, refunded, partially_refunded]
  expiresAt date // unpaid orders release their slots
}

Unavailability [icon: calendar-x, color: brown] {
  id string pk
  provider ref:Provider fk
  startDateTime date
  endDateTime date
  recurringMode enum // [none, daily, weekly, monthly]
  recurrenceEndDate date
  reason string
}

Booking [icon: book-check, color: red] {
  id string pk
  service ref:Service fk
  provider ref:Provider fk // assigned at order creation
  customer ref:User fk

  scheduledDate date
  startTime time
  endTime time
  
  status enum // [pending, confirmed, in_progress, completed, cancelled]
  cancelledBy enum // [customer, provider, manager, admin, system]
  cancelledReason string

  providerShare number // paise
  platformShare number // paise

  platformFeePercent number // snapshot taken from "platformSetting" at booking time

  payoutStatus enum // [pending, paid]
  payoutAt date
  
  price number // integer in paise
  order ref:Order fk
  region ref:Region fk
}

Payment [icon: credit-card, color: teal] {
  id string pk
  order ref:Order fk // unique
  totalAmount number // paise
  currency string

  razorpayOrderId string // unique
  razorpayPaymentId string
  razorpaySignature string
  refunds [{booking: ref:Booking, amount: number, razorpayRefundId: string, status: enum, createdAt: date}]
}

Review [icon: comments, color: yellow] {
  id string pk
  booking ref:Booking fk
  
  rating number
  review string
  
  provider ref:Provider fk
  service ref:Service fk
}

Region [icon: circle, color: green] {
  id string pk
  name string
  manager ref:Manager fk
  location 2dsphere
  isActive bool
}

PlatformSetting [icon: settings, color: blue] {
  platformFeePercent number
  maxAssignmentDistanceKm number
}
